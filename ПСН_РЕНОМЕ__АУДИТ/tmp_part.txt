Процедура ПриСозданииНаСервере(Отказ, СтандартнаяОбработка)
Параметры.Свойство("ЗаявкаСайт", ЗаявкаСайт);
Если ЗначениеЗаполнено(Объект.Ссылка) Тогда
	ЗаявкаСайт = Объект.ЗаявкаСайт;
КонецЕсли;
Если НЕ ЗначениеЗаполнено(ЗаявкаСайт) Тогда
	Элементы.ГруппаЗаявкаСайт.Видимость = Ложь;
КонецЕсли;
	
РаботаСИнтерфейсомКлиентСервер.УбратьКнопкуКоманднойПанелиФормы(ЭтаФорма.КоманднаяПанель, "ФормаУдалить");

ТекущийПользователь = ПараметрыСеанса.ТекущийПользователь;

ПараметрыТолькоПросмотр = Неопределено;
Параметры.Свойство("ТолькоПросмотр", ПараметрыТолькоПросмотр);
Если ПараметрыТолькоПросмотр = Неопределено Тогда
	ПараметрыТолькоПросмотр = Ложь;
КонецЕсли;

РаботаСИнтерфейсомСервер_УстановитьТолькоПросмотр = РаботаСИнтерфейсомСервер.УстановитьТолькоПросмотр(ЭтаФорма);
Если РаботаСИнтерфейсомСервер_УстановитьТолькоПросмотр Тогда 
	Элементы.Закрыто.ТолькоПросмотр = Истина;
Иначе
	Если НЕ НастройкиСервер.ЭтоАдминистратор() Тогда
		Элементы.Закрыто.ТолькоПросмотр = Истина;
	КонецЕсли;
КонецЕсли;
